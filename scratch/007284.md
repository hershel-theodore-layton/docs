Commit body:
Summary:
The semantic of `invoke_by_field_id` is that it can turn a runtime field id to compile-time id for a given thrift struct, e.g. `invoke_by_field_id<T>(10, f)` invokes `f(field_id<10>)` where `field_id<10>` is `std::integral_constant<FieldId, 10>`.

The current logic looks like this on high level

```
void invoke_by_field_id(id, callback) {
  if (id == field_id_1) { return callback(field_id<field_id_1>); }
  else if (id == field_id_2) { return callback(field_id<field_id_2>); }
  else if (id == field_id_3) { return callback(field_id<field_id_3>); }
  else if (id == field_id_4) { return callback(field_id<field_id_4>); }
  else if (id == field_id_5) { return callback(field_id<field_id_5>); }
  else if (id == field_id_6) { return callback(field_id<field_id_6>); }
  else if (id == field_id_7) { return callback(field_id<field_id_7>); }
  else if (id == field_id_8) { return callback(field_id<field_id_8>); }
  return invoke_by_field_id_2(id, callback);
}

void invoke_by_field_id_2(id, callback) {
  if (id == field_id_9) { return callback(field_id<field_id_9>); }
  else if (id == field_id_10) { return callback(field_id<field_id_10>); }
  else if (id == field_id_11) { return callback(field_id<field_id_11>); }
  else if (id == field_id_12) { return callback(field_id<field_id_12>); }
  else if (id == field_id_13) { return callback(field_id<field_id_13>); }
  else if (id == field_id_14) { return callback(field_id<field_id_14>); }
  else if (id == field_id_15) { return callback(field_id<field_id_15>); }
  else if (id == field_id_16) { return callback(field_id<field_id_16>); }
  return invoke_by_field_id_3(id, callback);
}
// ...
```

Compiler can automatically optimize if-else to switch, thus the optimized code looks like this

```
void invoke_by_field_id(id, callback) {
  switch (id) {
    case field_id_1: return callback(field_id<field_id_1>);
    case field_id_2: return callback(field_id<field_id_2>);
    case field_id_3: return callback(field_id<field_id_3>);
    case field_id_4: return callback(field_id<field_id_4>);
    case field_id_5: return callback(field_id<field_id_5>);
    case field_id_6: return callback(field_id<field_id_6>);
    case field_id_7: return callback(field_id<field_id_7>);
    case field_id_8: return callback(field_id<field_id_8>);
  }
  return invoke_by_field_id_2(id, callback);
}

void invoke_by_field_id_2(id, callback) {
  switch (id) {
    case field_id_9: return callback(field_id<field_id_9>);
    case field_id_10: return callback(field_id<field_id_10>);
    case field_id_11: return callback(field_id<field_id_11>);
    case field_id_12: return callback(field_id<field_id_12>);
    case field_id_13: return callback(field_id<field_id_13>);
    case field_id_14: return callback(field_id<field_id_14>);
    case field_id_15: return callback(field_id<field_id_15>);
    case field_id_16: return callback(field_id<field_id_16>);
  }
  return invoke_by_field_id_3(id, callback);
}

```

With FOLLY_ALWAYS_INLINE, compiler will collapse all batches first, then optimize to a single switch-case that is faster than multiple switch-cases

```
void invoke_by_field_id(id, callback) {
  switch (id) {
    case field_id_1: return callback(field_id<field_id_1>);
    case field_id_2: return callback(field_id<field_id_2>);
    case field_id_3: return callback(field_id<field_id_3>);
    case field_id_4: return callback(field_id<field_id_4>);
    case field_id_5: return callback(field_id<field_id_5>);
    case field_id_6: return callback(field_id<field_id_6>);
    case field_id_7: return callback(field_id<field_id_7>);
    case field_id_8: return callback(field_id<field_id_8>);
    case field_id_9: return callback(field_id<field_id_9>);
    case field_id_10: return callback(field_id<field_id_10>);
    case field_id_11: return callback(field_id<field_id_11>);
    // ...
  }
  return return callback(); // not found
}
```
