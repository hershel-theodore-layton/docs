Summary:
This diff implemented

```
template<class AnnotationType, class Struct, class Id>
const bool has_field_annotation();

// Build failure if Struct.Id doesn't have corresponding Annotation
template<class AnnotationType, class Struct, class Id>
const AnnotationType& get_field_annotation() { /* ... */ }
```

Potentially in the future, `has_field_annotation` can be constexpr if necessary.

Implementation wise, this diff defined two functions

```
namespace apache::thrift::detail::annotation {

template <class Struct, class Ident>
struct field_annotation_types;

template <class T>
const std::vector<std::any>& field_annotation_values(FieldId);
}
```

for the following thrift struct

```
struct MyStruct {
  Annotation1
  Annotation2
  1: i32 Field1;

  Annotation3
  Annotation4
  2: i32 Field2;
}
```

we will use code-gen to generate the following function

```
namespace apache::thrift::detail::annotation {

template <>
struct field_annotation_types<MyStruct, Field1> { using type = folly::tag_t<Annotation1, Annotation2>; };

template <>
struct field_annotation_types<MyStruct, Field2> { using type = folly::tag_t<Annotation3, Annotation4>; };

template <>
const std::vector<std::any>& field_annotation_values<MyStruct>(FieldId id) {
  switch (id) {
    case FieldId{1}: {
      const std::vector<std::any> ret = []{
        ret.push_back(/* annotation 1 */);
        ret.push_back(/* annotation 2 */);
      }();
      return ret;
    }
    case FieldId{2}: {
      const std::vector<std::any> ret = []{
        ret.push_back(/* annotation 3 */);
        ret.push_back(/* annotation 4 */);
      }();
      return ret;
    }
  }
  // unreachable
}
}
```

Which can be used by `has_field_annotation` and `get_field_annotation` to get the annotation value.
