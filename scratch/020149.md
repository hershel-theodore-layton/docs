Commit title: Add dsl::make_polymorphic_native_handle helper to describe class hierarchies
Commit body:
Summary:
This diff introduces a helper type alias, `dsl::make_polymorphic_native_handle`, to make it easy to describe polymorphic type hierarchies without code repetition.

In a future diff we use this in `t_whisker_generator` to represent Thrift's AST types' hierarchy.

This is what that looks like:
```
template <typename... Cases>
using handle_t = whisker::dsl::make_polymorphic_native_handle<Cases...>;

using interaction = handle_t<t_interaction>;
using service = handle_t<t_service>;
using interface = handle_t<t_interface, interaction, service>;

using function = handle_t<t_function>;
using stream = handle_t<t_stream>;
using sink = handle_t<t_sink>;
using include = handle_t<t_include>;
using package = handle_t<t_package>;

using program = handle_t<t_program>;

using list = handle_t<t_list>;
using set = handle_t<t_set>;
using map = handle_t<t_map>;
using container = handle_t<t_container, set, list, map>;

using const_ = handle_t<t_const>;
using enum_ = handle_t<t_enum>;
using enum_value = handle_t<t_enum_value>;
using exception = handle_t<t_exception>;
using field = handle_t<t_field>;
using primitive_type = handle_t<t_primitive_type>;

using exception = handle_t<t_exception>;
using union_ = handle_t<t_union>;
using throws = handle_t<t_throws>;
using paramlist = handle_t<t_paramlist>;
using struct_ = handle_t<t_struct, exception, union_, throws, paramlist>;
using structured = handle_t<t_structured, struct_>;
using typedef_ = handle_t<t_typedef>;
using type = handle_t<
    t_type,
    primitive_type,
    enum_,
    container,
    structured,
    typedef_,
    interface>;

using named =
    handle_t<t_named, const_, enum_value, field, function, type, program>;
using node = handle_t<t_node, named>;
```
