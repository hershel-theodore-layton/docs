The file docblock says it all.

This pattern is used several times in `coro/safe/Captures.h`, and it's way ugly if inlined, so I factored it out.

IMPORTANT: `Captures.h` includes this macro, and then promptly `#undef`s it. And then `CapturesTest.cpp` verifies that:

```cpp
#include <folly/coro/safe/Captures.h>
#include <folly/lang/Bindings.h>
#include <folly/portability/GTest.h>

#ifdef FOLLY_MOVABLE_AND_DEEP_CONST_LREF_COPYABLE
static_assert(false, "Leaked FOLLY_MOVABLE_AND_DEEP_CONST_LREF_COPYABLE macro");
#endif
```

For posterity, here's the content of the in-comment [Compiler Explorer link](https://godbolt.org/z/9bh4Wcso7) demonstrating **why** we resort to a macro instead of e.g. inheritance.

```cpp
#include <concepts>
#include <type_traits>

#define FOLLY_MOVABLE_AND_DEEP_CONST_LREF_COPYABLE(class_name, inner_type) \
  class_name(class_name&&) = default;                                      \
  class_name& operator=(class_name&&) = default;                           \
  class_name(class_name&)                                                  \
    requires std::is_lvalue_reference_v<inner_type>                        \
  = default;                                                               \
  class_name& operator=(class_name&)                                       \
    requires std::is_lvalue_reference_v<inner_type>                        \
  = default;                                                               \
  class_name(const class_name&)                                            \
    requires(std::is_lvalue_reference_v<inner_type> &&                     \
             std::is_const_v<std::remove_reference_t<inner_type>>)         \
  = default;                                                               \
  class_name& operator=(const class_name&)                                 \
    requires(std::is_lvalue_reference_v<inner_type> &&                     \
             std::is_const_v<std::remove_reference_t<inner_type>>)         \
  = default;                                                               \
  ~class_name() = default

// WISH: That leaf classes could just inherit from this, instead of needing
// to use the macro.
template <class Inner>
struct deep_const {
    deep_const() = default;
    FOLLY_MOVABLE_AND_DEEP_CONST_LREF_COPYABLE(deep_const, Inner);
};

template <class Inner>
struct capture : deep_const<Inner> {
    capture(Inner i) : p_(&i) {}
// UNFORTUNATELY: Removing any of this boilerplace causes compile errors
// So, the macro is overall the lesser evil.
#if 1
    capture(const capture&) = default;
    capture(capture&) = default;
    capture(capture&&) = default;
    capture& operator=(const capture&) = default;
    capture& operator=(capture&) = default;
    capture& operator=(capture&&) = default;
#endif

    const void* p_;
};

// DESIRED BEHAVIOR --

using intref = capture<int&>;
using cintref = capture<const int&>;

static_assert(!std::is_constructible_v<intref, const intref&>);
static_assert(std::is_nothrow_constructible_v<intref, intref&>);
static_assert(std::is_nothrow_constructible_v<intref, intref&&>);

static_assert(std::is_nothrow_constructible_v<cintref, const cintref&>);
static_assert(std::is_nothrow_constructible_v<cintref, cintref&>);
static_assert(std::is_nothrow_constructible_v<cintref, cintref&&>);

static_assert(!std::is_assignable_v<intref&, const intref&>);
static_assert(std::is_nothrow_assignable_v<intref&, intref&>);
static_assert(std::is_nothrow_assignable_v<intref&, intref&&>);

static_assert(std::is_nothrow_assignable_v<cintref&, const cintref&>);
static_assert(std::is_nothrow_assignable_v<cintref&, cintref&>);
static_assert(std::is_nothrow_assignable_v<cintref&, cintref&&>);

int main() {
    int i = 0;
    const cintref ii(i);
    cintref ij(i);
    ij = ii;
    return 0;
}
```
