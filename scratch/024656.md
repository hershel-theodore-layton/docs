This updates the codegen logic to account for type annotations on adapted fields. A struct that looks like this:
```
rust.Adapter{name = "::fbthrift_adapters::UuidAdapter<>"}
typedef binary uuid

struct Foo {
  rust.Type{name = "HashMap"}
  1: map<uuid, i32> bar;
  2: uuid baz;
  3: map<uuid, i32> biz;
}
```
would have this codegen:
```
pub type uuid = <::fbthrift_adapters::UuidAdapter<::std::vec::Vec<::std::primitive::u8>> as ::fbthrift::adapter::ThriftAdapter>::AdaptedType;

#[derive(Clone, PartialEq)]
pub struct Foo {
    pub bar: ::fbthrift::builtin_types::HashMap<crate::types::uuid, ::std::primitive::i32>,
    pub baz: <crate::types::adapters::uuid as ::fbthrift::adapter::ThriftAdapter>::AdaptedType,
    pub biz: <::fbthrift::adapter::MapMapAdapter<crate::types::adapters::uuid, ::fbthrift::adapter::IdentityAdapter<::std::primitive::i32>> as ::fbthrift::adapter::ThriftAdapter>::AdaptedType,
```
