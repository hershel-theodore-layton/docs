Commit title: Programmatic builder for Python type system
Commit body:
Summary:
Introduces a programmatic builder for constructing Python type system in code. This is the second milestone of the runtime type-system work.

Example:

```
I64 = PrimitiveTypeRef(Primitive.I64)
I32 = PrimitiveTypeRef(Primitive.I32)
STR = PrimitiveTypeRef(Primitive.STRING)
UNQ = PresenceQualifier.UNQUALIFIED
OPT = PresenceQualifier.OPTIONAL

ts = (
    TypeSystemBuilder()
    .add_enum(
        "meta.com/blog/Priority",
        [EnumValue("LOW", 0), EnumValue("MEDIUM", 1), EnumValue("HIGH", 2)],
    )
    # Opaque alias: give a primitive its own URI -- the IDL can't express this.
    .add_opaque_alias("meta.com/blog/Timestamp", I64)
    .add_struct(
        "meta.com/blog/User",
        [
            FieldSpec(FieldIdentity(1, "id"), UNQ, I64),
            FieldSpec(FieldIdentity(2, "name"), UNQ, STR),
            FieldSpec(FieldIdentity(3, "joined"), OPT, "meta.com/blog/Timestamp"),
        ],
    )
    .add_struct(
        "meta.com/blog/Comment",
        [
            FieldSpec(FieldIdentity(1, "author"), UNQ, "meta.com/blog/User"),
            FieldSpec(FieldIdentity(2, "text"), UNQ, STR),
            FieldSpec(FieldIdentity(3, "reply_to"), OPT, "meta.com/blog/Comment"),  # self-recursive
        ],
    )
    .add_struct(
        "meta.com/blog/Post",
        [
            FieldSpec(FieldIdentity(1, "author"), UNQ, "meta.com/blog/User"),
            FieldSpec(FieldIdentity(2, "priority"), UNQ, "meta.com/blog/Priority"),
            FieldSpec(FieldIdentity(3, "tags"), UNQ, SetTypeRef(STR)),
            FieldSpec(FieldIdentity(4, "comments"), UNQ, ListTypeRef(ref("meta.com/blog/Comment"))),
            FieldSpec(FieldIdentity(5, "reactions"), UNQ, MapTypeRef(STR, I32)),
        ],
        is_sealed=True,
    )
    .add_union(  # union fields are coerced to OPTIONAL
        "meta.com/blog/Notification",
        [
            FieldSpec(FieldIdentity(1, "mention"), UNQ, "meta.com/blog/User"),
            FieldSpec(FieldIdentity(2, "reply"), UNQ, "meta.com/blog/Comment"),
        ],
    )
    .build()
)

# Look up by URI; discriminate resolved edges with isinstance / match.
post = ts.get_user_defined_type_or_throw("meta.com/blog/Post")
match post.field_by_name("comments").type:
    case ListTypeRef(element_type=StructTypeRef(node=n)):
        print("comments is list of", n.uri)  # -> meta.com/blog/Comment
```
